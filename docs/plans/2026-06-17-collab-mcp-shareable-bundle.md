### Task 2: Bundle script `npm run bundle`

**Files:**
- Create: `scripts/bundle.mjs`
- Modify: root `package.json` (add `"bundle": "node scripts/bundle.mjs"`)
- Modify: `.gitignore` (add `dist-share/`)

**Interfaces — Produces:** `dist-share/collab-mcp/` (clean tree) + `dist-share/collab-mcp.zip`.

- [ ] **Step 1: Write `scripts/bundle.mjs`**

```js
// Assembles a clean, friend-installable bundle from the single source.
import { cpSync, rmSync, mkdirSync, writeFileSync, existsSync } from 'node:fs';
import { join } from 'node:path';
import { execSync } from 'node:child_process';

const ROOT = process.cwd();
const OUT = join(ROOT, 'dist-share', 'collab-mcp');

// EXCLUDE patterns (never ship). Applied as a filter during copy.
const DENY = [/(^|[\\/])collab\.db($|-)/, /(^|[\\/])node_modules([\\/]|$)/,
  /(^|[\\/])dist-share([\\/]|$)/, /(^|[\\/])\.env$/, /(^|[\\/])\.env\.[^/\\]+$/];
const deny = (p) => DENY.some((re) => re.test(p)) && !/\.env\.example$/.test(p);

function copyPkg(name) {
  cpSync(join(ROOT, name), join(OUT, name), { recursive: true,
    filter: (src) => !deny(src) });
}

rmSync(join(ROOT, 'dist-share'), { recursive: true, force: true });
mkdirSync(OUT, { recursive: true });

// 1. core + mcp source (built dist included if present; friend rebuilds anyway)
copyPkg('core');
copyPkg('mcp');

// 2. shared root configs
for (const f of ['tsconfig.base.json', '.nvmrc', '.gitignore', '.env.example'])
  if (existsSync(join(ROOT, f))) cpSync(join(ROOT, f), join(OUT, f));

// 3. onboarding assets (created in later tasks)
for (const f of ['README.bundle.md:README.md', 'LICENSE', 'AGENTS.md',
                 'scripts/seed-starter.mjs:scripts/seed-starter.mjs',
                 'skills/collab-workflow/SKILL.md:skills/collab-workflow/SKILL.md']) {
  const [src, dst] = f.includes(':') ? f.split(':') : [f, f];
  if (existsSync(join(ROOT, src))) {
    mkdirSync(join(OUT, dst, '..'), { recursive: true });
    cpSync(join(ROOT, src), join(OUT, dst));
  }
}

// 4. slim root package.json (workspaces = core + mcp only)
writeFileSync(join(OUT, 'package.json'), JSON.stringify({
  name: 'collab-mcp-bundle', private: true, type: 'module',
  engines: { node: '>=20.9.0' },
  workspaces: ['core', 'mcp'],
  scripts: {
    build: 'npm -w @collab-mcp/core run build && npm -w collab-mcp run build',
    'seed:starter': 'node scripts/seed-starter.mjs',
  },
}, null, 2) + '\n');

// 5. zip (PowerShell Compress-Archive is always present on Win)
try {
  execSync(`powershell -NoProfile -Command "Compress-Archive -Force -Path '${OUT}\\*' -DestinationPath '${join(ROOT, 'dist-share', 'collab-mcp.zip')}'"`,
    { stdio: 'inherit' });
} catch { console.warn('zip step skipped (Compress-Archive unavailable)'); }

console.log('Bundle ready at', OUT);
```
